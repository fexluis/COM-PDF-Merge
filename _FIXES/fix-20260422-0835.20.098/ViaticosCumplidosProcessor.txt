    /// <summary>
    /// Procesador encargado de generar los documentos de 'Cumplido' e 'Informe de Actividades'.

    internal class ViaticosCumplidosProcessor : IViaticosCumplidosProcessor

        private const int FILA_INICIAL = 4;
        private readonly IExcelRepository _excelRepository;
        private readonly IViaticosRepository _viaticosRepository;
        private readonly IUserService _userService;
        private readonly IPdfMetadataService _metadataService;

        /// <summary>

        public ViaticosCumplidosProcessor(

        /// <summary>
        /// Procesa por lotes todos los registros en la hoja de CONSULTAS que cumplan los criterios de fecha y estado.
        /// </summary>
        /// <param name="progressCallback">Opcional. NotificaciÃ³n de progreso.</param>
        /// <returns>NÃºmero de pares de documentos (CUM+INF) generados exitosamente.</returns>
        public int Procesar(Action<int, int> progressCallback = null)
        {
            dynamic wb = _excelRepository.GetActiveWorkbook();
            if (wb == null) return 0;
            dynamic excelApp = wb.Application;
            dynamic activeSheet = excelApp.ActiveSheet;

            if (!activeSheet.Name.Equals(ConstantsHelper.SHEET_CONSULTAS, StringComparison.OrdinalIgnoreCase))
            {
                MessageBox.Show("Seleccione la hoja de CONSULTAS", App.APP_NAME, MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return 0;
            }

            var config = _viaticosRepository.GetConfig();
            if (config == null) return 0;

            string folder = Path.Combine(config.CarpetaDestino, "Cumplidos");
            if (!Directory.Exists(folder)) Directory.CreateDirectory(folder);

            int ultimaFila = (int)activeSheet.Cells[activeSheet.Rows.Count, "A"].End[-4162 /*xlUp*/].Row;
            if (ultimaFila < FILA_INICIAL) return 0;

            int contador = 0;
            using (var scope = new ExcelAppStateHelper(excelApp))
            {
                scope.ApplyOptimized();
                object[,] data = _excelRepository.GetRangeValues(activeSheet, FILA_INICIAL, 1, ultimaFila, ConstantsHelper.CONSULTAS_COL_MAX);
                int totalRows = data.GetLength(0);
                var records = new List<ViaticoRow>();

                for (int i = 1; i <= totalRows; i++)
                {
                    var row = Map(data, i, FILA_INICIAL + i - 1);
                    if (ShouldProcess(row, config)) records.Add(row);
                }

                for (int i = 0; i < records.Count; i++)
                {
                    progressCallback?.Invoke(i + 1, records.Count);
                    if (ProcesarRegistro(wb, records[i], config, folder)) contador++;
                }
            }

            if (contador > 0) System.Diagnostics.Process.Start("explorer.exe", folder);
            return contador;
        }

        /// <summary>
        /// Procesa la fila de la hoja de CONSULTAS donde se encuentra el cursor del usuario.
        /// </summary>
        /// <param name="progressCallback">Callback para progreso (1/1).</param>
        /// <returns>1 si el registro se generÃ³ con Ã©xito, 0 en caso contrario.</returns>
        public int ProcesarFilaSeleccionada(Action<int, int> progressCallback = null)
        {
            dynamic wb = _excelRepository.GetActiveWorkbook();
            if (wb == null) return 0;
            dynamic excelApp = wb.Application;
            dynamic activeSheet = excelApp.ActiveSheet;

            if (!activeSheet.Name.Equals(ConstantsHelper.SHEET_CONSULTAS, StringComparison.OrdinalIgnoreCase))
            {
                MessageBox.Show("Seleccione la hoja de CONSULTAS", App.APP_NAME, MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return 0;
            }

            int rowIdx = excelApp.ActiveCell.Row;
            if (rowIdx < FILA_INICIAL)
            {
                MessageBox.Show("Seleccione una fila vÃ¡lida (4 o superior).", App.APP_NAME, MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return 0;
            }

            var config = _viaticosRepository.GetConfig();
            string folder = Path.Combine(config.CarpetaDestino, "Cumplidos");
            if (!Directory.Exists(folder)) Directory.CreateDirectory(folder);

            using (var scope = new ExcelAppStateHelper(excelApp))
            {
                scope.ApplyOptimized();
                progressCallback?.Invoke(1, 1);
                object[,] data = _excelRepository.GetRangeValues(activeSheet, rowIdx, 1, rowIdx, ConstantsHelper.CONSULTAS_COL_MAX);
                var row = Map(data, 1, rowIdx);

                if (!ShouldProcess(row, config))
                {
                    MessageBox.Show("La fila no cumple los criterios (Fecha, CDP, Estado).", App.APP_NAME, MessageBoxButtons.OK, MessageBoxIcon.Warning);
                    return 0;
                }

                if (ProcesarRegistro(wb, row, config, folder))
                {
                    System.Diagnostics.Process.Start("explorer.exe", folder);
                    return 1;
                }
            }
            return 0;
        }

        /// <summary>
        /// Mapea los datos de una fila de la matriz de Excel al modelo <see cref="ViaticoRow"/>.
        /// </summary>
        private ViaticoRow Map(object[,] data, int r, int excelIdx)
        {
            return new ViaticoRow
            {
                Index = excelIdx,
                NumeroComision = data[r, ConstantsHelper.CONSULTAS_NUMERO_COMISION]?.ToString() ?? "",
                FechaComision = ViaticosHelper.ToDateTime(data[r, ConstantsHelper.CONSULTAS_FECHA_COMISION]),
                FechaInicio = ViaticosHelper.ToDateTime(data[r, ConstantsHelper.CONSULTAS_FECHA_INICIO]),
                FechaFin = ViaticosHelper.ToDateTime(data[r, ConstantsHelper.CONSULTAS_FECHA_FIN]),
                Estado = data[r, ConstantsHelper.CONSULTAS_ESTADO]?.ToString()?.ToUpperInvariant() ?? "",
                Informe = data[r, ConstantsHelper.CONSULTAS_INFORME]?.ToString()?.ToUpperInvariant() ?? "",
                Cedula = data[r, ConstantsHelper.CONSULTAS_CEDULA]?.ToString() ?? "",
                NombreCompleto = data[r, ConstantsHelper.CONSULTAS_NOMBRE_COMPLETO]?.ToString() ?? "",
                Rol = data[r, ConstantsHelper.CONSULTAS_ROL]?.ToString() ?? "",
                Itinerario = data[r, ConstantsHelper.CONSULTAS_ITINERARIO]?.ToString()?.Trim() ?? "",
                RP = data[r, ConstantsHelper.CONSULTAS_RP]?.ToString()?.Trim() ?? "",
                CDP = data[r, ConstantsHelper.CONSULTAS_CDP]?.ToString()?.Trim() ?? "",
                OB = data[r, ConstantsHelper.CONSULTAS_OB]?.ToString()?.Trim() ?? "",
                OP = data[r, ConstantsHelper.CONSULTAS_OP]?.ToString()?.Trim() ?? "",
                ValorTotal = data[r, ConstantsHelper.CONSULTAS_VALOR_TOTAL]?.ToString() ?? "",
                ObjetoRAW = data[r, ConstantsHelper.CONSULTAS_OBJETO_RAW]?.ToString() ?? ""
            };
        }

        /// <summary>
        /// Determina si un registro especÃ­fico debe ser procesado aplicando criterios de negocio y filtros.
        /// </summary>
        private bool ShouldProcess(ViaticoRow row, ViaticosConfig config)
        {
            if (row.FechaInicio < config.FechaInicio || row.FechaInicio > config.FechaFin) return false;
            if (!string.IsNullOrEmpty(config.Cdp) && !row.CDP.Equals(config.Cdp, StringComparison.OrdinalIgnoreCase)) return false;
            if (row.Estado == "REVOCADA" || row.Estado == "RECHAZADA" || row.Estado == "ANULADA") return false;
            if (row.Informe == "VERIFICADO" || row.Informe == "REVOCADA" || row.Informe == "REINTEGRO") return false;
            if (string.IsNullOrEmpty(row.Cedula)) return false;
            return true;
        }

        private bool ProcesarRegistro(dynamic wb, ViaticoRow row, ViaticosConfig config, string folder)
        {
            try
            {
                var emp = _viaticosRepository.GetEmpleado(row.Cedula);
                if (emp != null)
                {
                    row.NombreCompleto = emp.NombreCompleto?.ToUpperInvariant() ?? row.NombreCompleto.ToUpperInvariant();
                    row.Rol = emp.Rol ?? row.Rol;
                }
                row.ObjetoProcesado = ExtraerDespuesPagoViaticos(row.ObjetoRAW);

                GenerateFile(wb, row, config, folder, "CUMPLIDO", "CUM");
                GenerateFile(wb, row, config, folder, "INFORME", "INF");
                return true;
            }
            catch (Exception ex)
            {
                LoggerHelper.Error($"Error registro {row.Cedula}", ex);
                return false;
            }
        }

        /// <summary>
        /// Orquesta la creaciÃ³n de un archivo individual basÃ¡ndose en una plantilla (CUM o INF).
        /// Exporta el resultado a PDF e inyecta los metadatos correspondientes.
        /// </summary>
        private void GenerateFile(dynamic wb, ViaticoRow row, ViaticosConfig config, string folder, string template, string prefix)
        {
            if (prefix == "INF") LoggerHelper.Info($"[INF] Iniciando generaciÃ³n para fila {row.Index}: Viatico {row.NumeroComision}");

            string sheetName = $"{prefix}-{row.Index}";
            dynamic ws = CopiarPlantilla(wb, sheetName, template);
            
            if (prefix == "CUM") LlenarCumplido(ws, row, config);
            else
            {
                LoggerHelper.Info($"[INF] Llenando datos del informe para {row.Cedula}");
                LlenarInforme(ws, row, config);
            }

            string fileName = $"{row.NumeroComision}-{row.Cedula}-{prefix}-{LimpiarNombre(row.NombreCompleto)}";
            if (fileName.Length > 40) fileName = fileName.Substring(0, 40);

            if (prefix == "INF") LoggerHelper.Info($"[INF] Guardando archivos XLSX/PDF: {fileName}");
            Guardar(wb.Application, ws, fileName, folder);
            
            // Guardar metadatos en el PDF generado
            try
            {
                string fullPdfPath = Path.Combine(folder, fileName + ".pdf");
                string fullUser = _userService.GetUsuarioWindows();
                var parts = fullUser.Split('\\');
                string cleanUser = parts.Length > 1 ? parts[1].ToUpper() : fullUser.ToUpper();

                _metadataService.GuardarMetadatos(fullPdfPath, new Metadata
                {
                    Titulo = fileName,
                    Autor = cleanUser,
                    Asunto = row.ObjetoProcesado,
                    PalabrasClave = $"{row.Cedula}, {row.NombreCompleto}, {config.FullDependencia}",
                    Fecha = row.FechaInicio,
                    MetaDocumento = prefix == "CUM" ? TipoMetadata.VIA_CUMPLIDO : TipoMetadata.VIA_INFORME,
                    CDP = row.CDP,
                    RP = row.RP,
                    OB = row.OB,
                    OP = row.OP,
                    Tipo = prefix == "CUM" ? "Cumplido" : "Informe",
                    Comision = row.NumeroComision
                });
            }
            catch (Exception ex)
            {
                LoggerHelper.Error($"Error al guardar metadatos para {fileName}", ex);
            }
            
            EliminarHoja(wb, sheetName);
            if (prefix == "INF") LoggerHelper.Success($"[INF] Informe generado correctamente");
        }

        private dynamic CopiarPlantilla(dynamic wb, string name, string template)
        {
            dynamic wsT = wb.Worksheets[template];
            wsT.Visible = -1;
            wsT.Copy(Type.Missing, wb.Worksheets[wb.Worksheets.Count]);
            dynamic wsN = wb.Application.ActiveSheet;
            wsN.Name = name;
            
            try { wsN.Unprotect(); } catch { }

            wsT.Visible = 2;
            return wsN;
        }

        /// <summary>
        /// Llenado fÃ­sico de las celdas para la plantilla de 'Cumplido'.
        /// Genera el texto narrativo integrando datos de la comisiÃ³n y firma.
        /// </summary>
        private void LlenarCumplido(dynamic ws, ViaticoRow row, ViaticosConfig config)
        {
            ws.Range["A15"].Value = $"Que el seÃ±or {row.NombreCompleto}, identificado con la cÃ©dula de ciudadanÃ­a No.{row.Cedula}, {TextoRol(row.Rol)}, cumpliÃ³ con el desplazamiento autorizado mediante No. SIIF NaciÃ³n {row.NumeroComision} del {row.FechaComision:dd/MM/yyyy}, a la ciudad de {row.Itinerario}, durante el perÃ­odo comprendido entre el {row.FechaInicio:dd/MM/yyyy} y el {row.FechaFin:dd/MM/yyyy}, con el objeto de {row.ObjetoProcesado}";
            ws.Range["A17"].Value = $"Para constancia, se firma en Cali, a los {FechaATexto(row.FechaInicio.AddDays(5))}.";
            ws.Range["A26"].Value = "Nombre: " + config.NombreDT;
            ws.Range["A27"].Value = "Cargo: " + config.CargoDT;

            ws.Calculate();
        }

        /// <summary>
        /// Llenado fÃ­sico de las celdas para la plantilla de 'Informe de Actividades'.
        /// Mapea campos de SIIF NaciÃ³n y personales del comisionado.
        /// </summary>
        private void LlenarInforme(dynamic ws, ViaticoRow row, ViaticosConfig config)
        {
            DateTime f = row.FechaInicio.AddDays(5);
            ws.Range["D7"].Value = f.Day; ws.Range["E7"].Value = f.Month; ws.Range["F7"].Value = f.Year;
            ws.Range["B19"].Value = row.FechaInicio.Day; ws.Range["C19"].Value = row.FechaInicio.Month; ws.Range["D19"].Value = row.FechaInicio.Year;
            ws.Range["F19"].Value = row.FechaFin.Day; ws.Range["G19"].Value = row.FechaFin.Month; ws.Range["H19"].Value = row.FechaFin.Year;
            ws.Range["D8"].Value = row.NombreCompleto;
            ws.Range["D10"].Value = row.Cedula;
            ws.Range["D12"].Value = config.FullDependencia;
            ws.Range["M7"].Value = "SIIF NaciÃ³n " + row.NumeroComision;
            ws.Range["M10"].Value = row.RP;
            //ws.Range["M12"].Value = config.NombreDT?.ToUpperInvariant();
            ws.Range["J19"].Value = row.Itinerario?.ToUpperInvariant();
            ws.Range["B26"].Value = row.ObjetoProcesado?.ToUpperInvariant();
            ws.Range["N28"].Value = row.ValorTotal.Replace("$", "").Replace(",", "").Replace(".00", "").Trim();

            ws.Calculate();
        }

        private string TextoRol(string r)
        {
            r = r?.ToUpperInvariant()?.Trim() ?? "";
            if (r.StartsWith("DOC")) return "Docente cÃ¡tedra";
            if (r.StartsWith("FUN")) return "Funcionario";
            if (r.StartsWith("CON")) return "Contratista";
            if (r.StartsWith("ESA")) return "Contratista ESA";
            return "Docente";
        }

        private string FechaATexto(DateTime d)
        {
            string[] dias = { "", "uno", "dos", "tres", "cuatro", "cinco", "seis", "siete", "ocho", "nueve", "diez", "once", "doce", "trece", "catorce", "quince", "diecisÃ©is", "diecisiete", "dieciocho", "diecinueve", "veinte", "veintiuno", "veintidÃ³s", "veintitrÃ©s", "veinticuatro", "veinticinco", "veintisÃ©is", "veintisiete", "veintiocho", "veintinueve", "treinta", "treinta y uno" };
            string[] meses = { "", "enero", "febrero", "marzo", "abril", "mayo", "junio", "julio", "agosto", "septiembre", "octubre", "noviembre", "diciembre" };
            return $"{dias[d.Day]} ({d.Day}) dÃ­as del mes de {meses[d.Month]} del aÃ±o {d.Year}";
        }

        /// <summary>
        /// Limpia el texto original del objeto de la comisiÃ³n utilizando expresiones regulares para extraer solo la descripciÃ³n central.
        /// Elimina redundancias como "PAGO VIATICOS", "PARA", y aÃ±os de vigencia.
        /// </summary>
        /// <param name="s">Cadena de texto original.</param>
        /// <returns>Texto limpio y conciso.</returns>
        private string ExtraerDespuesPagoViaticos(string s)
        {
            if (string.IsNullOrWhiteSpace(s)) return string.Empty;

            string vigenciaSuffix = App.APP_VIGENCIA.Substring(App.APP_VIGENCIA.Length - 2);

            // Pattern to handle misspellings of VIATICOS
            string viaticosPattern = @"(?:VIATICOS|VITICOS|VIATICA|VIATICO|VAITICOS|VAITICO|VIATCIOS)";

            // Pattern: PAGO [VIATICOS variant] followed by (optional number and PARA) OR (optional specific number ending in vigenciaSuffix)
            string pattern = $@"(?i)PAGO\s+{viaticosPattern}(?:(?:\s+\d+)?\s+PARA|(?:\s+\d*{vigenciaSuffix})?)\b";

            var match = Regex.Match(s, pattern);
            if (match.Success)
            {
                return s.Substring(match.Index + match.Length).Trim();
            }
            return s.Trim();
        }

        private string LimpiarNombre(string s) => Regex.Replace(s?.Trim() ?? "", @"\s+", "-");

        private void Guardar(dynamic app, dynamic ws, string name, string folder)
        {
            string xls = Path.Combine(folder, name + ".xlsx");
            string pdf = Path.Combine(folder, name + ".pdf");
            ws.Copy();
            dynamic wb = app.ActiveWorkbook;
            dynamic wsN = wb.Worksheets[1];            

            app.DisplayAlerts = false;
            try
            {
                if (File.Exists(xls)) File.Delete(xls);

                if (File.Exists(ConstantsHelper.THEME_PATH))
                {
                    try { wb.Theme.ThemeColorScheme.Load(ConstantsHelper.THEME_PATH); }
                    catch (Exception ex) { LoggerHelper.Warning("No se pudo cargar el tema: " + ex.Message); }
                }

                wb.SaveAs(xls, 51);
                App.ConfigurarPagina(wsN, true);
                wsN.ExportAsFixedFormat(0, pdf);
            }
            catch (Exception ex)
            {
                LoggerHelper.Error($"[Guardar] Fallo al guardar {name}: {ex.Message}", ex);
                throw;
            }
            finally
            {
                wb.Close(false);
                app.DisplayAlerts = true;
            }
        }

        private void EliminarHoja(dynamic wb, string name)
        {
            wb.Application.DisplayAlerts = false;
            try { wb.Worksheets[name].Delete(); } catch { }
            wb.Application.DisplayAlerts = true;
        }
