using Xunit;
using FluentAssertions;
using System;
using System.Collections.Generic;
using ConversorPDF.XLL.Core.Helpers;
using ConversorPDF.XLL.Core.Models.Academic;

namespace ConversorPDF.Tests.Unit
{
    public class GridToSesionMapperTests
    {
        // â”€â”€â”€ Helpers de construcciÃ³n de DTOs â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

        private static GridRowDto FilaMarcada(int idTercero = 1) => new GridRowDto
        {
            IsChecked = true, IsNewRow = false,
            Tag = new Sesion { IdTercero = idTercero },
            Horas = 8, Duracion = 8,
            FechaInicial = new DateTime(2026, 4, 23),
            FechaFinal = new DateTime(2026, 4, 23)
        };

        private static GridRowDto FilaMarcadaSinTag() => new GridRowDto
        { IsChecked = true, IsNewRow = false, Tag = null };

        private static GridRowDto FilaDesmarcada() => new GridRowDto
        { IsChecked = false, IsNewRow = false, Tag = new Sesion { IdTercero = 1 } };

        private static GridRowDto FilaNueva() => new GridRowDto
        { IsChecked = false, IsNewRow = true, Tag = null };

        // â”€â”€â”€ TEST CRÃTICO C-01a: Reproductor del bug real â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€

        [Fact]
        public void Extract_6FilasMarcadasCorrectamente_Retorna6Sesiones()
        {
            var rows = new List<GridRowDto>();
            for (int i = 0; i < 6; i++) rows.Add(FilaMarcada());
            rows.Add(FilaNueva()); 

            var resultado = GridToSesionMapper.Extract(rows, idTerceroMaster: 1, idVinculacion: 1);

            resultado.Should().HaveCount(6);
        }

        // El test anterior era invÃ¡lido. La integridad se dispara si hay un desfase entre
        // lo que el usuario ve marcado y lo que el motor de extracciÃ³n acepta.
        // En la implementaciÃ³n actual, los errores de validaciÃ³n disparan excepciones inmediatamente
        // dentro del loop, por lo que el chequeo final de conteo es una red de seguridad secundaria.

        [Fact]
        public void Extract_UltimaFilaMarcadaSinTag_LanzaExcepcionConNumeroDeFila()
        {
            var rows = new List<GridRowDto> { FilaMarcada(), FilaMarcada(), FilaMarcadaSinTag() };

            Action act = () => GridToSesionMapper.Extract(rows, idTerceroMaster: 1, idVinculacion: 1);

            act.Should().Throw<InvalidOperationException>()
               .WithMessage("*Fila 3*")
               .And.Message.Should().Contain("modelo asociado");
        }

        [Fact]
        public void Extract_FilasConIsNewRow_SonIgnoradas()
        {
            var rows = new List<GridRowDto>
            {
                FilaMarcada(), FilaMarcada(), FilaNueva() 
            };

            var resultado = GridToSesionMapper.Extract(rows, idTerceroMaster: 1, idVinculacion: 1);

            resultado.Should().HaveCount(2);
        }

        [Fact]
        public void Extract_GridVacio_RetornaListaVacia()
        {
            var rows = new List<GridRowDto>();

            var resultado = GridToSesionMapper.Extract(rows, idTerceroMaster: 1, idVinculacion: 1);

            resultado.Should().BeEmpty();
        }

        [Fact]
        public void Extract_DocenteDesincronizado_LanzaExcepcion()
        {
            var rows = new List<GridRowDto> { FilaMarcada(idTercero: 2) }; 

            Action act = () => GridToSesionMapper.Extract(rows, idTerceroMaster: 99, idVinculacion: 1);

            act.Should().Throw<InvalidOperationException>()
               .WithMessage("*docente de esta sesiÃ³n no coincide*");
        }
    }
