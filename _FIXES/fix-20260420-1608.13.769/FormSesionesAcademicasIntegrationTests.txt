using Microsoft.VisualStudio.TestTools.UnitTesting;
using System;
using System.Collections.Generic;
using System.Windows.Forms;
using System.Threading;
using System.Linq;
using System.Globalization;
using ConversorPDF.XLL.FormAcademic;
using ConversorPDF.XLL.Core.Models.Academic;

namespace ConversorPDF.Tests.UI
{
    [TestClass]
    public class FormSesionesAcademicasIntegrationTests
    {
        private FormSesionesAcademicas _form;
        private DataGridView _grid;

        [TestInitialize]
        public void Setup()
        {
            Exception threadEx = null;
            var thread = new Thread(() =>
            {
                try
                {
                    _form = new FormSesionesAcademicas();
                    _grid = TestHelper.GetPrivateField<DataGridView>(_form, "dgvSesiones");
                }
                catch (Exception ex) { threadEx = ex; }
            });
            thread.SetApartmentState(ApartmentState.STA);
            thread.Start();
            thread.Join();
            if (threadEx != null) throw new AssertInconclusiveException($"Setup fallÃ³: {threadEx.Message}");
        }

        private void AgregarFilaAlGrid(bool marcada, Sesion tag = null)
        {
            _form.Invoke((Action)(() => {
                int rowIndex = _grid.Rows.Add();
                var row = _grid.Rows[rowIndex];
                row.Cells["ColChecked"].Value = marcada;
                row.Cells["ColHoras"].Value = 8;
                row.Cells["ColDuracion"].Value = 8;
                row.Cells["ColInicio"].Value = "23/04/2026";
                row.Cells["ColFinal"].Value = "23/04/2026";
                row.Tag = tag;
            }));
        }

        [TestMethod]
        public void PrepararSesiones_ConteoCoincide_RetornaCantidadCorrecta()
        {
            AgregarFilaAlGrid(marcada: true, tag: new Sesion { IdTercero = 1, Horas = 8 });
            AgregarFilaAlGrid(marcada: true, tag: new Sesion { IdTercero = 1, Horas = 8 });
            AgregarFilaAlGrid(marcada: false, tag: new Sesion { IdTercero = 1, Horas = 8 });

            var resultado = TestHelper.InvokePrivateMethod<IEnumerable<Sesion>>(_form, "PrepararSesionesDesdeGrid");
            var lista = resultado.ToList();

            Assert.AreEqual(2, lista.Count);
        }

        [TestCleanup]
        public void Cleanup()
        {
            _form?.Invoke((Action)(() => _form.Dispose()));
        }
    }

    public static class TestHelper
    {
        public static T GetPrivateField<T>(object obj, string fieldName)
        {
            var field = obj.GetType().GetField(fieldName, System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance);
            if (field == null) {
                // Probablemente autofiltro o nombre diferente? Intentar con property si no es field
                var prop = obj.GetType().GetProperty(fieldName, System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance);
                return (T)prop.GetValue(obj);
            }
            return (T)field.GetValue(obj);
        }

        public static T InvokePrivateMethod<T>(object obj, string methodName, params object[] args)
        {
            var method = obj.GetType().GetMethod(methodName, System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance);
            return (T)method.Invoke(obj, args);
        }
    }
}
